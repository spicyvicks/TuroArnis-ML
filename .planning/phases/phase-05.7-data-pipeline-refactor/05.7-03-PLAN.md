    Extract metrics and compare with front model:
    
    ```python
    import json
    
    # Load left model history
    with open('hybrid_classifier/logs/history_left.json') as f:
        left = json.load(f)
    
    # Load front model for comparison
    with open('hybrid_classifier/logs/history_front.json') as f:
        front = json.load(f)
    
    left_best = max(left['val_acc'])
    left_final_train = left['train_acc'][-1]
    left_gap = left_final_train - left_best
    front_best = max(front['val_acc'])
    
    print(f"LEFT MODEL:")
    print(f"  Best Val Acc: {left_best:.1%}")
    print(f"  Final Train Acc: {left_final_train:.1%}")
    print(f"  Train-Test Gap: {left_gap:+.1%}")
    print(f"  Epochs: {len(left['val_acc'])}")
    print(f"\nComparison with FRONT:")
    print(f"  Front Best: {front_best:.1%}")
    print(f"  Gap (Front - Left): {front_best - left_best:+.1%}")
    
    # Determine status
    if left_best >= 0.70:
        status = "PASS"
    elif left_best >= 0.60:
        status = "RETRY"
    else:
        status = "INVESTIGATE"
    
    print(f"\nSTATUS: {status}")
    ```
    
    Report the status classification for checkpoint.
